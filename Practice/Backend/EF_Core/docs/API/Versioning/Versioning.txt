Versioning
  Versioning is to have multiple stages of our web API application.

    - Using NuGet package Versioning
      1- Install: 
          ASP.Versioning.Http
          ASP.Versioning.Mvc.ApiExplorer

      2- In Program.cs:
        // APIVersioning package
        builder.Services.AddApiVersioning(options =>
        {
            options.AssumeDefaultVersionWhenUnspecified = true;
        });

        Above we are accessing the default version of an API controller method (version 1)

      3. The controller

      using Microsoft.AspNetCore.Mvc;

      using WebApiVersioning.Models.DTOs;

      namespace WebApiVersioning.Controllers
      {
          [Route("api/[controller]")]
          [ApiController]
          [ApiVersion("1.0")]
          [ApiVersion("2.0")]
          public class CountriesController : ControllerBase
          {
              // Version 1
              [MapToApiVersion(version: "1.0")]
              [HttpGet]
              public IActionResult GetAllCountries()
              {
                  var countriesDomainModel = CountriesData.GetCountries();

                  // Convert to DTO
                  var response = new List<CountryDTOV1>();

                  foreach (var country in countriesDomainModel)
                  {
                      response.Add(new CountryDTOV1
                      {
                          Id = country.Id,
                          Name = country.Name
                      });
                  }
                  return Ok(response);
              }

              // Version 2
              [MapToApiVersion(version: "2.0")]
              [HttpGet]
              public IActionResult GetAllCountriesV2()
              {
                  var countriesDomainModel = CountriesData.GetCountries();

                  // Convert to DTO
                  var response = new List<CountryDTOV2>();

                  foreach (var country in countriesDomainModel)
                  {
                      response.Add(new CountryDTOV2
                      {
                          Id = country.Id,
                          CountryName = country.Name
                      });
                  }
                  return Ok(response);
              }
          }
      }

    [Route("api/[controller]")] 
      Will run the HTTP Web API request for that controller.
    
    [ApiVersion("1.0")]
    [ApiVersion("2.0")] 
      Includes the versions available of the API.

    [MapToApiVersion] 
      Specifies which version should this action method run.

    4. In SwaggerUI, or in Postman:
    Add a new HTTP GET request for GetAllCountries action method:

    Return the default API version  (1.0)
    https://localhost:7134/api/Countries 

    Return API version  (1.0)
    https://localhost:7134/api/Countries?api-version=1.0

    Return the default API version  (2.0)
    https://localhost:7134/api/Countries?api-version=2.0



    Force the client to specify a version
      - For that to happen, we need to integrate a version inside the [Route] attribute
        by adding /v{version:apiVersion}

      [Route("api/v{version:apiVersion}/[controller]")]


      In SwaggerUI/Postman:
      https://localhost:7134/api/v1/Countries  // returns v1
      https://localhost:7134/api/v2/Countries  // returns v1
